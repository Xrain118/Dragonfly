cmake_minimum_required(VERSION 3.16)
project(dragonfly_flight)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(dragonfly_physics SHARED src/physics.cpp)
target_compile_features(dragonfly_physics PUBLIC cxx_std_17)
target_link_libraries(dragonfly_physics gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS dragonfly_physics DESTINATION lib)
ament_python_install_package(dragonfly_flight)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/flight scripts/keyboard scripts/verify DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(flight_core test/test_core.py ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1" "PYTHONNOUSERSITE=1")
  ament_add_pytest_test(flight_keyboard test/test_keyboard.py ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1" "PYTHONNOUSERSITE=1")
  ament_add_pytest_test(flight_clearance test/test_clearance.py ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1" "PYTHONNOUSERSITE=1")
  ament_add_pytest_test(flight_runtime test/test_runtime.py ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1" "PYTHONNOUSERSITE=1")
  ament_add_pytest_test(flight_run test/test_run.py ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1" "PYTHONNOUSERSITE=1")
endif()
ament_package()
